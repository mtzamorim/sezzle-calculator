package main

import (
        "context"
        "errors"
        "log/slog"
        "net"
        "net/http"
        "os"
        "os/signal"
        "path/filepath"
        "strings"
        "syscall"
        "time"

        "github.com/matheusAmorim123/sezzle-calculator/backend/internal/httpapi"
)

func main() {
        logger := slog.New(slog.NewTextHandler(os.Stdout, nil))
        port := os.Getenv("PORT")
        if port == "" {
                port = "8080"
        }

        api := httpapi.NewHandler()
        handler := api
        if root := os.Getenv("STATIC_DIR"); root != "" {
                handler = withFrontend(api, root)
        }
        server := &http.Server{
                Addr:              ":" + port,
                Handler:           handler,
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      15 * time.Second,
                IdleTimeout:       60 * time.Second,
        }

        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()
        listener, err := net.Listen("tcp", server.Addr)
        if err != nil {
                logger.Error("listen failed", "error", err)
                os.Exit(1)
        }
        logger.Info("calculator ready", "address", listener.Addr())
        if err := serve(ctx, server, listener); err != nil {
                logger.Error("server failed", "error", err)
                os.Exit(1)
        }
}

// Shutdown makes Serve return immediately. Keep the main goroutine alive until
// in-flight requests have drained, instead of exiting as soon as Serve returns.
func serve(ctx context.Context, server *http.Server, listener net.Listener) error {
        stopped := make(chan error, 1)
        go func() { stopped <- server.Serve(listener) }()
        select {
        case err := <-stopped:
                if errors.Is(err, http.ErrServerClosed) {
                        return nil
                }
                return err
        case <-ctx.Done():
                shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
                defer cancel()
                if err := server.Shutdown(shutdownCtx); err != nil {
                        _ = server.Close()
                        return err
                }
                if err := <-stopped; !errors.Is(err, http.ErrServerClosed) {
                        return err
                }
                return nil
        }
}

// In the optional Docker image, the same Go server serves the built React app.
// API paths never fall through to index.html.
func withFrontend(api http.Handler, root string) http.Handler {
        files := http.FileServer(http.Dir(root))
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if r.URL.Path == "/api" || strings.HasPrefix(r.URL.Path, "/api/") {
                        api.ServeHTTP(w, r)
                        return
                }
                if r.Method != http.MethodGet && r.Method != http.MethodHead {
                        http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
                        return
                }
                w.Header().Set("X-Content-Type-Options", "nosniff")
                path := filepath.Join(root, filepath.Clean("/"+r.URL.Path))
                if info, err := os.Stat(path); err == nil && !info.IsDir() {
                        files.ServeHTTP(w, r)
                        return
                }
                // Missing JS/CSS must be a 404, never a successful HTML response.
                if filepath.Ext(r.URL.Path) != "" || strings.HasPrefix(r.URL.Path, "/assets/") {
                        http.NotFound(w, r)
                        return
                }
                http.ServeFile(w, r, filepath.Join(root, "index.html"))
        })
}

// Package calculator contains the pure arithmetic rules and their validation.
// It has no knowledge of HTTP, JSON, or persistence.
package calculator

import (
        "math"
)

// Operation lists the supported arithmetic operations.
type Operation string

const (
        Add        Operation = "add"
        Subtract   Operation = "subtract"
        Multiply   Operation = "multiply"
        Divide     Operation = "divide"
        Power      Operation = "power"
        SquareRoot Operation = "sqrt"
        Percent    Operation = "percent"
)

// Input uses pointers so that missing JSON operands can be distinguished from 0.
type Input struct {
        Operation Operation `json:"operation"`
        A         *float64  `json:"a"`
        B         *float64  `json:"b,omitempty"`
}

// Failure is an expected domain error. The transport layer maps Code to HTTP.
type Failure struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

func (e *Failure) Error() string { return e.Message }

func invalid(message string) error   { return &Failure{Code: "invalid_input", Message: message} }
func undefined(message string) error { return &Failure{Code: "math_error", Message: message} }

// Calculate evaluates an operation. Percentage means "a percent of b":
// e.g. (percent, 15, 200) = 30. Square root accepts only a.
func Calculate(in Input) (float64, error) {
        switch in.Operation {
        case Add, Subtract, Multiply, Divide, Power, SquareRoot, Percent:
        default:
                return 0, &Failure{Code: "unsupported_operation", Message: "Unsupported operation."}
        }

        if in.A == nil {
                return 0, invalid("The first operand (a) is required.")
        }
        a := *in.A
        if math.IsNaN(a) || math.IsInf(a, 0) {
                return 0, invalid("The first operand (a) must be a finite number.")
        }

        if in.Operation == SquareRoot {
                if in.B != nil {
                        return 0, invalid("Square root accepts only the first operand (a).")
                }
                if a < 0 {
                        return 0, undefined("Cannot calculate the square root of a negative number.")
                }
                return math.Sqrt(a), nil
        }

        if in.B == nil {
                return 0, invalid("The second operand (b) is required.")
        }
        b := *in.B
        if math.IsNaN(b) || math.IsInf(b, 0) {
                return 0, invalid("The second operand (b) must be a finite number.")
        }

        var result float64
        switch in.Operation {
        case Add:
                result = a + b
        case Subtract:
                result = a - b
        case Multiply:
                result = a * b
        case Divide:
                if b == 0 {
                        return 0, undefined("Division by zero is not allowed.")
                }
                result = a / b
        case Power:
                result = math.Pow(a, b)
        case Percent:
                result = a / 100 * b
        }

        if math.IsNaN(result) || math.IsInf(result, 0) {
                return 0, undefined("The result is not a finite real number.")
        }
        return result, nil
}

// Package httpapi exposes the calculator through a small JSON REST API.
package httpapi

import (
        "encoding/json"
        "errors"
        "io"
        "mime"
        "net/http"
        "strings"

        "github.com/matheusAmorim123/sezzle-calculator/backend/internal/calculator"
)

const maxRequestBody = 16 << 10 // 16 KiB, including trailing whitespace.

type calculationResponse struct {
        Operation calculator.Operation `json:"operation"`
        A         float64              `json:"a"`
        B         *float64             `json:"b,omitempty"`
        Result    float64              `json:"result"`
}

type errorResponse struct {
        Error *calculator.Failure `json:"error"`
}

// NewHandler returns an HTTP handler suitable for httptest or http.Server.
func NewHandler() http.Handler {
        mux := http.NewServeMux()
        mux.HandleFunc("/api/v1/health", allowMethods([]string{"GET", "HEAD"}, func(w http.ResponseWriter, _ *http.Request) {
                writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
        }))
        mux.HandleFunc("/api/v1/calculate", allowMethods([]string{"POST"}, calculate))
        mux.HandleFunc("/", func(w http.ResponseWriter, _ *http.Request) {
                writeError(w, http.StatusNotFound, "not_found", "The requested API endpoint does not exist.")
        })
        return mux
}

func allowMethods(allowed []string, next http.HandlerFunc) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                for _, method := range allowed {
                        if r.Method == method {
                                next(w, r)
                                return
                        }
                }
                w.Header().Set("Allow", strings.Join(allowed, ", "))
                writeError(w, http.StatusMethodNotAllowed, "method_not_allowed", "The HTTP method is not supported for this endpoint.")
        }
}

func calculate(w http.ResponseWriter, r *http.Request) {
        mediaType, _, err := mime.ParseMediaType(r.Header.Get("Content-Type"))
        if err != nil || mediaType != "application/json" {
                writeError(w, http.StatusUnsupportedMediaType, "unsupported_media_type", "Content-Type must be application/json.")
                return
        }
        r.Body = http.MaxBytesReader(w, r.Body, maxRequestBody)
        decoder := json.NewDecoder(r.Body)
        decoder.DisallowUnknownFields()

        var input *calculator.Input
        if err := decoder.Decode(&input); err != nil {
                writeDecodeError(w, err)
                return
        }
        if input == nil {
                writeError(w, http.StatusBadRequest, "invalid_json", "Request body must be a JSON object.")
                return
        }
        // Check the second read too: an otherwise valid object followed by more than
        // 16 KiB of whitespace is still an oversized request, not a different 400 error.
        if err := decoder.Decode(&struct{}{}); err != io.EOF {
                writeDecodeError(w, err)
                return
        }
        result, err := calculator.Calculate(*input)
        if err != nil {
                var failure *calculator.Failure
                if errors.As(err, &failure) {
                        status := http.StatusBadRequest
                        if failure.Code == "math_error" {
                                status = http.StatusUnprocessableEntity
                        }
                        writeJSON(w, status, errorResponse{Error: failure})
                        return
                }
                writeError(w, http.StatusInternalServerError, "internal_error", "Unable to complete the calculation.")
                return
        }
        writeJSON(w, http.StatusOK, calculationResponse{Operation: input.Operation, A: *input.A, B: input.B, Result: result})
}

func writeDecodeError(w http.ResponseWriter, err error) {
        var tooLarge *http.MaxBytesError
        if errors.As(err, &tooLarge) {
                writeError(w, http.StatusRequestEntityTooLarge, "payload_too_large", "The request body exceeds 16 KiB.")
                return
        }
        writeError(w, http.StatusBadRequest, "invalid_json", "Request body must contain exactly one JSON object with numeric operands.")
}

func writeError(w http.ResponseWriter, status int, code, message string) {
        writeJSON(w, status, errorResponse{Error: &calculator.Failure{Code: code, Message: message}})
}

func writeJSON(w http.ResponseWriter, status int, body any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.Header().Set("Cache-Control", "no-store")
        w.Header().Set("X-Content-Type-Options", "nosniff")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(body)
}
